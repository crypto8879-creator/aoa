SETUP:

1. Run command:
   npm create vue@latest my-vue-p1

2. Terminal Prompts:
   Select No to all options (TypeScript, Router, Pinia, Vitest, ESLint, Prettier).

3. cd my-vue-p1
4. npm install
5. npm run dev
   (Runs on http://localhost:5173)

6. Files:
   - Create src/users.json
   - Replace src/App.vue with code below.





COMPONENTS:


COMPONENT: users.json
PATH: src/users.json

[
  { "username": "admin", "role": "Admin", "permissions": ["Create", "Edit", "Delete"] },
  { "username": "editor", "role": "Editor", "permissions": ["Create", "Edit"] },
  { "username": "viewer", "role": "Viewer", "permissions": ["Read Only"] }
]





COMPONENT: App.vue
PATH: src/App.vue

<script setup>
import { ref } from 'vue'
import users from './users.json'

const username = ref('')
const password = ref('')
const currentUser = ref(null)
const error = ref(false)

function handleLogin() {
  const found = users.find(u => u.username === username.value && password.value === '123')
  if (found) {
    currentUser.value = found
    error.value = false
    username.value = ''
    password.value = ''
  } else {
    error.value = true
  }
}

function handleLogout() {
  currentUser.value = null
}
</script>

<template>
  <div style="font-family: sans-serif; max-width: 400px; margin: 30px auto; padding: 20px; border: 1px solid #ccc; border-radius: 8px;">
    <!-- 1. v-if / v-else: Toggle between Login Screen and Dashboard -->
    <div v-if="!currentUser">
      <h2>System Login</h2>
      <!-- 2. v-model: Two-way input binding -->
      <p><input v-model="username" placeholder="Username (admin/editor/viewer)" style="width: 100%; padding: 8px;" /></p>
      <p><input v-model="password" type="password" placeholder="Password (123)" style="width: 100%; padding: 8px;" /></p>
      
      <!-- 3. v-show: Conditional visibility without DOM removal -->
      <p v-show="error" style="color: red;">Invalid credentials! (Use password: 123)</p>
      
      <!-- 4. v-on (@): Event listening -->
      <button @click="handleLogin" style="padding: 8px 16px; background: #2563eb; color: white; border: none; border-radius: 4px; cursor: pointer;">
        Login
      </button>
    </div>

    <div v-else>
      <h2>Welcome, {{ currentUser.username }}</h2>
      <p>Role: <strong>{{ currentUser.role }}</strong></p>
      <button @click="handleLogout" style="padding: 6px 12px; cursor: pointer;">Logout</button>

      <h3>Allowed Permissions:</h3>
      <ul>
        <!-- 5. v-for: List iteration with key binding -->
        <li v-for="(perm, index) in currentUser.permissions" :key="index">
          ✅ {{ perm }}
        </li>
      </ul>
    </div>
  </div>
</template>
